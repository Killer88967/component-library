## Usage

```tsx
import { Switch } from "@component-library/ui";

<Switch id="airplane-mode" />
<Switch size="sm" />
```

## CSS only

A native checkbox with `role="switch"`:

```html
<input type="checkbox" role="switch" class="cl-switch" id="airplane-mode" />
<input type="checkbox" role="switch" class="cl-switch" data-size="sm" />
```
