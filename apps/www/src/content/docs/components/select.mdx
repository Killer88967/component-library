## Usage

```tsx
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@component-library/ui";

<Select>
  <SelectTrigger className="w-48">
    <SelectValue placeholder="Theme" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="light">Light</SelectItem>
    <SelectItem value="dark">Dark</SelectItem>
  </SelectContent>
</Select>;
```

## CSS only

The CSS-only version is a real `<select>`, styled like [Native Select](/docs/components/native-select). It works in forms with no JavaScript and opens the system picker on phones and tablets.
