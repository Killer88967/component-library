## Usage

```tsx
import { InputGroup, InputGroupAddon, InputGroupInput } from "@component-library/ui";

<InputGroup>
  <InputGroupInput placeholder="Search..." />
  <InputGroupAddon>
    <SearchIcon />
  </InputGroupAddon>
</InputGroup>;
```

`InputGroupAddon` takes an `align` of `inline-start` (default), `inline-end`, `block-start` or `block-end`.
