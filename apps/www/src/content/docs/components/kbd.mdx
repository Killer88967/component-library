## Usage

```tsx
import { Kbd, KbdGroup } from "@component-library/ui";

<KbdGroup>
  <Kbd>Ctrl</Kbd>
  <Kbd>K</Kbd>
</KbdGroup>;
```

## CSS only

```html
<kbd class="cl-kbd-group">
  <kbd class="cl-kbd">Ctrl</kbd>
  <kbd class="cl-kbd">K</kbd>
</kbd>
```
